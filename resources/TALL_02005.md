# ScriptJob管理器

```python
import maya.cmds as cmds


class SingletonMeta(type):
    _instances = {}

    def __call__(cls, *args, **kwargs):
        if cls not in cls._instances:
            cls._instances[cls] = super().__call__(*args, **kwargs)
        return cls._instances[cls]


class ScriptJobManagerBase(metaclass=SingletonMeta):
    registration_script_job_list: list = []
    executed_job_list = []

    condition_and_event_parameters = [
        'attributeAdded',
        'attributeChange',
        'attributeDeleted',
        'conditionChange',
        'conditionFalse',
        'conditionTrue',
        'connectionChange',
        'event',
        'nodeDeleted',
        'nodeNameChanged',
        'optionVarChanged',
        'uiDeleted',
    ]

    @classmethod
    def add_job(cls, **kwargs):
        cls.registration_script_job_list.append(kwargs)

    @classmethod
    def run_jobs(cls):
        for i in cls.registration_script_job_list:
            cls.executed_job_list.append(cmds.scriptJob(**i))

    @classmethod
    def kill_jobs(cls):
        for idx, i in enumerate(cls.executed_job_list):
            cmds.scriptJob(kill=i)
            cls.executed_job_list.pop(idx)

    @classmethod
    def add_script_job_decorator(cls, *decorator_args, **decorator_kwargs):
        def decorator(func):
            # 保存原始函数的引用（只保存一次）
            original = getattr(func, '__original_func__', func)
            func.__original_func__ = original

            # 在函数声明时立即注册，不论堆叠多少次，都使用原始函数
            for condition, lst in decorator_kwargs.items():
                if condition in cls.condition_and_event_parameters:
                    lst.append(original)
            cls.registration_script_job_list.append(decorator_kwargs)

            def wrapper(*args, **kwargs):
                return func(*args, **kwargs)

            # 让返回的 wrapper 也保持原始函数引用
            wrapper.__original_func__ = original
            return wrapper

        return decorator


@ScriptJobManagerBase.add_script_job_decorator(event=['SceneOpened'])
@ScriptJobManagerBase.add_script_job_decorator(event=['NewSceneOpened'])
def onSceneOpened(**kwargs):
    print(kwargs)


ScriptJobManagerBase.add_job(event=['SelectionChanged', onSceneOpened])

sjmb = ScriptJobManagerBase()

sjmb.run_jobs()

sjmb.kill_jobs()

for i in ScriptJobManagerBase.registration_script_job_list:
    print(i)

for i in ScriptJobManagerBase.executed_job_list:
    print(i)

cmds.scriptJob(listJobs=True)
```
