import dataclasses

from typing import Optional
import maya.cmds as cmds


@dataclasses.dataclass
class _VtxInfluencesDC:
    idx: int
    influences: int


@dataclasses.dataclass
class _MeshInfluencesDC:
    name: str
    vtx_influences: list[_VtxInfluencesDC] = dataclasses.field(default_factory=list)

    def get_max_influence(self):
        return max((i.influences for i in self.vtx_influences), default=0)

    def get_min_influence(self):
        return min((i.influences for i in self.vtx_influences), default=0)

    def get_vertices_greater_than_number(self, number: int):
        return [
            f'{self.name}.vtx[{i.idx}]'
            for i in self.vtx_influences
            if i.influences > number
        ]

    def is_mesh_influences_greater_than_number(self, number: int):
        '''
        判断mesh的影响骨骼数量是否大于number
        :param number:
        :return:
        '''
        return any(i.influences > number for i in self.vtx_influences)

    def get_transform_name(self):
        if cmds.nodeType(self.name) == 'mesh':
            return cmds.listRelatives(self.name, parent=True)[0]


def get_influences_bulk(vertices, skin_cluster, all_influences, ignore_below=0.00001):
    influences_per_vertex = []
    for vertex in vertices:
        # 单独查询每个顶点的权重
        weights = cmds.skinPercent(skin_cluster, vertex, query=True, value=True, ignoreBelow=ignore_below)
        influences = [inf for inf, weight in zip(all_influences, weights) if weight > 0]
        influences_per_vertex.append(influences)
    return influences_per_vertex


def get_mesh_influences(mesh_name: str) -> Optional[_MeshInfluencesDC]:
    vertices = cmds.ls(mesh_name + '.vtx[*]', flatten=True)
    skin_clusters = cmds.ls(cmds.listHistory(mesh_name), type='skinCluster')

    if not skin_clusters:
        return None

    skin_cluster = skin_clusters[0]
    all_influences = cmds.skinCluster(skin_cluster, query=True, inf=True)
    influences_per_vertex = get_influences_bulk(vertices, skin_cluster, all_influences)

    vtx_influences_list = (
        _VtxInfluencesDC(idx, len(influences))
        for idx, influences in enumerate(influences_per_vertex)
    )
    return _MeshInfluencesDC(mesh_name, list(vtx_influences_list))


def get_meshes_influences(mesh_transform_name_list: list[str]):
    '''
    获取所有mesh的影响骨骼信息。
    :param mesh_transform_name_list:
    :return:
    '''
    return [
        influences
        for mesh in mesh_transform_name_list
        if cmds.objExists(mesh) and cmds.nodeType(mesh) == 'mesh'
        if (influences := get_mesh_influences(mesh))
    ]


def get_vertexes_influences_greater_than_max_num(mesh_list, max_num=4):
    '''
    获取所有顶点的影响骨骼数量大于max_num的顶点。
    场景中的Verts数量为40万时，大概需要13秒进行处理。
    i7-12700K 3.61GHz

    :param mesh_list:
    :param max_num:
    :return:
    '''
    return [
        vertex
        for mesh_influence in get_meshes_influences(mesh_list)
        for vertex in mesh_influence.get_vertices_greater_than_number(max_num)
    ]


def get_meshes_influences_greater_than_max_num(mesh_list, max_num=4):
    '''
    返回mesh中任意vertex的influence大于指定数字的mesh数据列表。
    '''
    return [
        mesh_influence
        for mesh_influence in get_meshes_influences(mesh_list)
        if mesh_influence.is_mesh_influences_greater_than_number(max_num)
    ]
